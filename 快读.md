char fast_getchar() {
	static char buf[1 << 20];
	static char *pos = buf, *end = buf;
	if (pos == end) {
		end = buf + fread(buf, 1, sizeof(buf), stdin);
		pos = buf;
		if (pos == end) return EOF;
	}
	return *pos++;
}
