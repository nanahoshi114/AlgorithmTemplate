struct Node {
    int l, r;
    mutable int num;
    bool operator<(const Node &r) const { return l < r.l; }
};
std::set<Node> odt;
auto split(int mid) {
    auto it = odt.lower_bound({mid + 1, 0, 0});
    if (it != odt.end() && it->l == mid + 1) {
        return it;
    }
    --it;
    int l = it->l, r = it->r, num = it->num;
    odt.erase(it);
    odt.insert({l, mid, num});
    auto [res, _] = odt.insert({mid + 1, r, num});
    return res;
}
void update(int l, int r, int num) {
    auto itr = split(r), itl = split(l - 1);
    odt.erase(itl, itr);
    odt.insert({l, r, num});
}
