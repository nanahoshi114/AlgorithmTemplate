auto toposort(){
    vector<int> ans;
    queue<int> S;
    for(int i = 1; i <= n; i++){
        if(!ent[i]) S.push(i);
    }
    while(!S.empty()){
        int now = S.front();
        S.pop();
        ans.push_back(now);
        for(int i = head[now]; ~i; i = edge[i].next){
            int to = edge[i].to;
            if(!(--ent[to])){
                S.push(to);
            }
        }
    }
    return ans;
}
